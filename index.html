site_root <- function(session) {
  cd <- session$clientData
  path <- sub("(index_files/.*)?app_[^/]+/?$", "", isolate(cd$url_pathname))
  port <- isolate(cd$url_port)
  paste0(isolate(cd$url_protocol), "//", isolate(cd$url_hostname),
         if (!is.null(port) && nzchar(port)) paste0(":", port), path)
}

fetch_file <- function(rel, root) {
  if (!file.exists(rel)) {
    d <- dirname(rel)
    if (d != ".") dir.create(d, recursive = TRUE, showWarnings = FALSE)
    tmp <- paste0(rel, ".part")
    download.file(paste0(root, rel), tmp, mode = "wb", quiet = TRUE)
    file.rename(tmp, rel)
  }
  rel
}

load_network <- function(kind, root) {
  f <- if (kind == "leaf") {
    c("09_TOM_UMAP_embedding_leaf.csv", "09_top_edges_leaf.csv", "leaf_network_maizeminedescription.txt")
  } else {
    c("09_TOM_UMAP_embedding_underground.csv", "09_top_edges_underground.csv", "und_network_maizeminedescription.txt")
  }
  umap  <- read.csv(fetch_file(f[1], root), stringsAsFactors = FALSE)
  edges <- read.csv(fetch_file(f[2], root), stringsAsFactors = FALSE)
  desc  <- read.delim(fetch_file(f[3], root), stringsAsFactors = FALSE, check.names = FALSE)
  names(desc)[1:2] <- c("gene_id_Zm", "OG")
  desc$Description <- as.character(desc[[grep("^description", names(desc))[1]]])
  desc$sink_source <- if ("sink_source" %in% names(desc)) as.character(desc$sink_source) else NA_character_

  og_desc <- desc %>%
    group_by(OG) %>%
    summarise(
      Zea_mays_genes = paste(unique(gene_id_Zm), collapse = ", "),
      Description = {
        d <- unique(Description[!is.na(Description) & Description != ""])
        if (length(d) == 0) "No description available" else paste(d, collapse = " / ")
      },
      .groups = "drop"
    )
  umap <- umap %>%
    left_join(og_desc %>% select(OG, Zea_mays_genes), by = c("gene" = "OG")) %>%
    mutate(Zea_mays_genes = ifelse(is.na(Zea_mays_genes), "None", Zea_mays_genes))
  coords <- umap %>% select(gene, UMAP1, UMAP2)
  edges <- edges %>%
    left_join(coords, by = c("from" = "gene")) %>% rename(x0 = UMAP1, y0 = UMAP2) %>%
    left_join(coords, by = c("to" = "gene")) %>% rename(x1 = UMAP1, y1 = UMAP2) %>%
    filter(!is.na(x0), !is.na(x1))
  modules <- umap %>% distinct(module_label, color, module_num) %>% arrange(module_num)
  list(umap = umap, edges = edges, desc = desc, og_desc = og_desc, modules = modules,
       plot_dir = if (kind == "leaf") "module_plots" else "module_plots_underground")
}

load_atlas <- function(root, progress = function(value, detail) NULL) {
  if (isTRUE(DATA$ready)) return(invisible(TRUE))
  progress(0.05, "sample metadata")
  md <- as.data.frame(readRDS(fetch_file("metadata.rds", root)))
  md <- md %>%
    transmute(sample_id, species, accession, tissue, tissue_type = as.character(tissue_type),
              season = as.character(season), week = as.integer(week),
              julian_days = as.integer(julian_days), date = as.Date(date_formatted), life_history)
  progress(0.15, "photosynthetic index")
  ps <- read.csv(fetch_file("ps_index.csv", root), stringsAsFactors = FALSE)
  md$ps_index <- ps$ps_index[match(md$sample_id, ps$sample_id)]
  DATA$metadata <- md
  DATA$avail <- md %>% distinct(species, accession, season, tissue_type)
  DATA$acc2sp <- with(distinct(md, accession, species), setNames(species, accession))

  progress(0.25, "gene → orthogroup map")
  gm <- readRDS(fetch_file("mapping_full_xz.rds", root))
  DATA$gene2og <- setNames(gm$Orthogroup, gm$GeneID)

  progress(0.55, "expression index")
  DATA$expr_index <- read.csv(fetch_file("expr_shards/index.csv", root), stringsAsFactors = FALSE)
  DATA$samples <- readLines(fetch_file("expr_shards/samples.txt", root))
  DATA$gene_choices <- c(sort(unique(gm$GeneID[gm$Species == "Zea_mays"])), DATA$expr_index$OG)

  progress(0.65, "weather")
  w <- read.csv(fetch_file("weather_daily.csv", root))
  w$season <- as.character(w$season)
  DATA$weather <- w %>%
    arrange(season, julian_day) %>%
    group_by(season) %>%
    mutate(cum_gdd = cumsum(ifelse(julian_day >= 200 & !is.na(gdd_C), gdd_C, 0))) %>%
    ungroup()

  progress(0.75, "co-expression networks")
  DATA$leaf <- load_network("leaf", root)
  DATA$und  <- load_network("underground", root)
  DATA$ready <- TRUE
  invisible(TRUE)
}

# One orthogroup = one row in a uint16 shard (VST x 1000); see data_prep/build_expression_shards.py
get_expression <- function(og, root) {
  i <- match(og, DATA$expr_index$OG)
  if (is.na(i)) return(NULL)
  f <- fetch_file(sprintf("expr_shards/shard_%03d.bin", DATA$expr_index$shard[i]), root)
  n <- length(DATA$samples)
  raw <- readBin(f, "raw", n = file.size(f))
  off <- DATA$expr_index$row[i] * n * 2
  v <- readBin(raw[(off + 1):(off + 2 * n)], "integer", n = n, size = 2, signed = FALSE, endian = "little")
  data.frame(sample_id = DATA$samples, expression = v / 1000, stringsAsFactors = FALSE)
}

season_tag <- function(s) paste(sort(unique(s)), collapse = ", ")

tissue_choices <- function(seasons) {
  a <- DATA$avail %>% filter(season %in% seasons)
  out <- lapply(tissue_groups, function(tt) {
    tt <- tt[tt %in% a$tissue_type]
    if (!length(tt)) return(NULL)
    setNames(tt, sapply(tt, function(t)
      paste0(tissue_labels[[t]], " · ", season_tag(a$season[a$tissue_type == t]))))
  })
  Filter(Negate(is.null), out)
}

species_choices <- function(seasons, tissues) {
  a <- DATA$avail %>% filter(season %in% seasons, tissue_type %in% tissues)
  sp <- sort(unique(DATA$avail$species))
  setNames(sp, sapply(sp, function(s) {
    x <- a$season[a$species == s]
    if (length(x)) paste0(pretty_name(s), " · ", season_tag(x))
    else paste0(pretty_name(s), " \u00b7 no data for this season/tissue")
  }))
}

accession_choices <- function(accs) {
  a <- DATA$avail %>% filter(accession %in% accs)

    setNames(ac, sapply(ac, function(x) paste0(x, " · ", season_tag(a$season[a$accession == x]))))

}

weather_by_week <- function(var, window) {
  w <- DATA$weather
  if (window == "calendar") {
    w %>%
      filter(julian_day >= 200) %>%
      mutate(week = floor((julian_day - 196) / 7)) %>%
      group_by(season, week) %>%
      summarise(value = mean(.data[[var]], na.rm = TRUE), .groups = "drop")
  } else {
    d <- DATA$metadata %>% distinct(season, week, julian_days)
    d$value <- mapply(function(s, j) {
      x <- w[w$season == s & w$julian_day >= j - 6 & w$julian_day <= j, ]
      mean(x[[var]], na.rm = TRUE)
    }, d$season, d$julian_days)
    d %>% group_by(season, week) %>% summarise(value = mean(value, na.rm = TRUE), .groups = "drop")
  }
}

}

abbrev_species <- function(g) {
  parts <- strsplit(g, "_")[[1]]
  if (length(parts) < 2) return(pretty_name(g))
  paste0(substr(parts[1], 1, 1), ". ", paste(parts[-1], collapse = " "))
}
group_label <- function(g, group_var) {
  if (group_var == "species") paste0("<i>", abbrev_species(g), "</i>") else pretty_name(g)
}
# Push labels apart vertically so direct labels never overlap
spread_labels <- function(y, gap) {
  o <- order(y); ys <- y[o]
  if (length(ys) > 1) for (i in 2:length(ys)) ys[i] <- max(ys[i], ys[i - 1] + gap)
  out <- numeric(length(y)); out[o] <- ys; out
}

make_timecourse <- function(df, s, group_var, y_rng, wx = NULL, wx_var = NULL, highlight = "", file_stub = "timecourse") {
  d <- df %>% filter(season == s)
  sm <- d %>%
    group_by(week, group = as.character(.data[[group_var]])) %>%
    summarise(mean_expr = mean(expression, na.rm = TRUE),
              se_expr = if (n() > 1) sd(expression, na.rm = TRUE) / sqrt(n()) else 0,
              n = n(), .groups = "drop") %>%
    arrange(group, week)
  pal <- palette_for(group_var)
  groups <- unique(sm$group)
  hl <- if (!is.null(highlight) && nzchar(highlight) && highlight %in% groups) highlight else NULL
  direct <- length(groups) <= 8
  bars <- !is.null(wx) && identical(wx_var, "gdd_C")
  ex_ax <- if (bars) "y2" else "y"     # GDD bars must sit on the primary axis to draw underneath
  wx_ax <- if (bars) "y" else "y2"

  wk <- DATA$metadata %>% filter(season == s) %>%
    group_by(week) %>% summarise(d = min(date), .groups = "drop") %>% arrange(week)
  ticktext <- paste0("w", wk$week, "<br><span style='color:#999'>", format(wk$d, "%b %d"), "</span>")
  mon <- format(wk$d, "%b")
  runs <- rle(mon); ends <- cumsum(runs$lengths); starts <- ends - runs$lengths + 1
  shapes <- lapply(seq_along(runs$values), function(i) list(
    type = "rect", xref = "x", yref = "paper", layer = "below",
    x0 = wk$week[starts[i]] - 0.5, x1 = wk$week[ends[i]] + 0.5, y0 = 0, y1 = 1,
    fillcolor = unname(month_fill[runs$values[i]] %||% "rgba(0,0,0,0.03)"), line = list(width = 0)))
  annotations <- lapply(seq_along(runs$values), function(i) list(
    x = mean(c(wk$week[starts[i]], wk$week[ends[i]])), y = 1, xref = "x", yref = "paper",
    yanchor = "bottom", text = runs$values[i], showarrow = FALSE, font = list(size = 11, color = "#777")))

  p <- plot_ly()

  # ---- weather layer 1: GDD bars (underneath everything) ----
  wd <- NULL
  if (!is.null(wx)) {
    lim <- range(wx$value[is.finite(wx$value)])
    wd <- wx %>% filter(season == s, week %in% wk$week, is.finite(value)) %>% arrange(week)
    if (bars && nrow(wd) > 0) {
      p <- p %>% add_bars(x = wd$week, y = wd$value, yaxis = wx_ax, width = 0.55,
                          marker = list(color = "rgba(232,163,61,0.30)", line = list(color = "rgba(168,112,26,0.55)", width = 1)),
                          name = weather_short[[wx_var]], hoverinfo = "text", showlegend = !direct,
                          text = paste0("Week ", wd$week, "<br>Mean daily GDD: ", round(wd$value, 1), " °C"),
                          textposition = "none")
    }
  }

  # ---- expression: context groups first, highlighted group last (on top) ----
  ord <- c(setdiff(groups, hl), hl)
  for (g in ord) {
    gd <- sm[sm$group == g, ]
    base <- if (g %in% names(pal)) unname(pal[g]) else "#333333"
    focus <- is.null(hl) || g == hl
    col <- if (focus) base else GREY_CTX
    if (focus) {
      p <- p %>% add_polygons(x = c(gd$week, rev(gd$week)),
                              y = c(gd$mean_expr + gd$se_expr, rev(gd$mean_expr - gd$se_expr)),
                              yaxis = ex_ax, fillcolor = col, opacity = 0.16, line = list(color = "transparent"),
                              legendgroup = g, showlegend = FALSE, hoverinfo = "skip", name = pretty_name(g))
    }
    p <- p %>% add_trace(x = gd$week, y = gd$mean_expr, yaxis = ex_ax, type = "scatter", mode = "lines+markers",
                         line = list(color = col, width = if (!is.null(hl) && g == hl) 3.5 else if (focus) 2.5 else 1.5),
                         marker = list(color = col, size = if (focus) 6 else 4),
                         name = pretty_name(g), legendgroup = g, showlegend = !direct, hoverinfo = "text",
                         text = paste0("<b>", pretty_name(g), "</b><br>Week ", gd$week, "<br>Mean VST: ",
                                       round(gd$mean_expr, 2), " ± ", round(gd$se_expr, 2), " (n = ", gd$n, ")"))
  }

  # Direct labels at the end of each line (instead of a legend)
  x_max <- max(wk$week) + 0.6
  if (direct && length(groups)) {
    last <- sm %>% group_by(group) %>% filter(week == max(week)) %>% ungroup()
    gap <- diff(y_rng) * 0.075
    last$y_lab <- last$mean_expr
    at_end <- last$week == max(last$week)
    last$y_lab[at_end] <- spread_labels(last$mean_expr[at_end], gap)
    for (i in seq_len(nrow(last))) {
      g <- last$group[i]
      focus <- is.null(hl) || g == hl
      annotations[[length(annotations) + 1]] <- list(
        x = last$week[i] + 0.18, y = last$y_lab[i], xref = "x", yref = ex_ax, xanchor = "left",
        text = group_label(g, group_var), showarrow = FALSE,
        font = list(size = 11, color = if (focus) (if (g %in% names(pal)) unname(pal[g]) else "#333") else "#aaaaaa"))
    }
    x_max <- max(wk$week) + 3
  }

  # ---- weather layer 2: temperature as a dashed line with triangles ----
  wx_layout <- NULL
  if (!is.null(wd) && nrow(wd) > 0) {
    lab <- weather_axis[[wx_var]]
    if (bars) {
      wx_layout <- axis_style(title = list(text = lab, font = list(color = GDD_INK)), side = "right",
                              range = c(0, lim[2] * 1.25), tickfont = list(color = GDD_INK, size = 11),
                              linecolor = GDD_INK, tickcolor = GDD_INK)
    } else {
      pad <- diff(lim) * 0.08
      p <- p %>% add_trace(x = wd$week, y = wd$value, yaxis = wx_ax, type = "scatter", mode = "lines+markers",
                           line = list(color = TEMP_INK, width = 1.8, dash = "dash"),
                           marker = list(symbol = "triangle-up", size = 13, color = temp_colour(wd$value, lim), opacity = 1,
                                         line = list(color = TEMP_INK, width = 1.2)),
                           name = weather_short[[wx_var]], showlegend = !direct, hoverinfo = "text",
                           text = paste0("Week ", wd$week, "<br>", lab, ": ", round(wd$value, 1)))
      wx_layout <- axis_style(title = list(text = paste("▲", lab), font = list(color = TEMP_INK)),
                              overlaying = "y", side = "right", range = c(min(lim[1], -2) - pad, lim[2] + pad),
                              tickfont = list(color = TEMP_INK, size = 11), linecolor = TEMP_INK, tickcolor = TEMP_INK)
      shapes[[length(shapes) + 1]] <- list(type = "line", xref = "paper", x0 = 0, x1 = 1, yref = "y2", y0 = 0, y1 = 0,
                                           line = list(color = "#9ecae1", dash = "dot", width = 1.5))
      annotations[[length(annotations) + 1]] <- list(xref = "paper", x = 0.01, yref = "y2", y = 0, xanchor = "left",
                                                     yanchor = "bottom", text = "0 °C", showarrow = FALSE,
                                                     font = list(color = "#4a90c2", size = 10))
      # direct label on the weather series, at its first point (left side is free of expression labels)

# ---- Photosynthetic-index axis (1 = photosynthetic -> 0 = senescent) ----
ps_fits <- function(d, span = 0.5) {
  bind_rows(lapply(split(d, d$group), function(gd) {
    if (nrow(gd) < 8 || length(unique(gd$ps_index)) < 5) return(NULL)
    fit <- tryCatch(suppressWarnings(loess(expression ~ ps_index, data = gd, span = span)), error = function(e) NULL)
    if (is.null(fit)) return(NULL)
    grid <- seq(min(gd$ps_index), max(gd$ps_index), length.out = 60)
    pr <- tryCatch(suppressWarnings(predict(fit, data.frame(ps_index = grid), se = TRUE)), error = function(e) NULL)
    if (is.null(pr)) return(NULL)
    data.frame(group = gd$group[1], ps_index = grid, fit = pr$fit, se = pr$se.fit,
               ci_low = pr$fit - 1.96 * pr$se.fit, ci_high = pr$fit + 1.96 * pr$se.fit, n_libraries = nrow(gd))
  }))
}

ps_bands <- function() {
  cols <- colorRamp(c("#4c9946", "#8cc052", "#c9d048", "#f6d654"))(seq(0, 1, length.out = 10))
  lapply(1:10, function(i) list(
    type = "rect", xref = "x", yref = "paper", layer = "below", x0 = 1 - (i - 1) / 10, x1 = 1 - i / 10,
    y0 = 0, y1 = 1, line = list(width = 0),
    fillcolor = sprintf("rgba(%d,%d,%d,0.07)", round(cols[i, 1]), round(cols[i, 2]), round(cols[i, 3]))))
}

make_ps_curve <- function(df, group_var, y_rng, highlight = "", file_stub = "ps_index") {
  d <- df %>% filter(!is.na(ps_index)) %>% mutate(group = as.character(.data[[group_var]]))
  pal <- palette_for(group_var)
  groups <- sort(unique(d$group))
  hl <- if (!is.null(highlight) && highlight %in% groups) highlight else NULL
  direct <- length(groups) <= 8
  fits <- ps_fits(d)
  colour_of <- function(g) if (g %in% names(pal)) unname(pal[g]) else "#333333"
  ord <- c(setdiff(groups, hl), hl)

  p <- plot_ly()
  for (g in ord) {   # libraries as faint context
    pts <- d[d$group == g, ]
    focus <- is.null(hl) || g == hl
    p <- p %>% add_trace(type = "scatter", mode = "markers", x = pts$ps_index, y = pts$expression,
                         marker = list(color = if (focus) colour_of(g) else GREY_CTX, size = 4,
                                       opacity = if (focus) 0.3 else 0.15),
                         legendgroup = g, showlegend = FALSE, hoverinfo = "text",
                         text = paste0(pts$sample_id, "<br>", pretty_name(pts$accession), " · ", pts$season,
                                       " · week ", pts$week, "<br>PS index: ", round(pts$ps_index, 2),
                                       "<br>VST: ", round(pts$expression, 2)))
  }
  annotations <- list()
  if (!is.null(fits) && nrow(fits)) {
    for (g in intersect(ord, unique(fits$group))) {
      f <- fits[fits$group == g, ]
      focus <- is.null(hl) || g == hl
      col <- if (focus) colour_of(g) else GREY_CTX
      if (focus) p <- p %>% add_polygons(x = c(f$ps_index, rev(f$ps_index)), y = c(f$ci_high, rev(f$ci_low)),
                                         fillcolor = col, opacity = 0.16, line = list(color = "transparent"),
                                         legendgroup = g, showlegend = FALSE, hoverinfo = "skip")
      p <- p %>% add_trace(type = "scatter", mode = "lines", x = f$ps_index, y = f$fit,
                           line = list(color = col, width = if (!is.null(hl) && g == hl) 3.5 else if (focus) 2.8 else 1.5),
                           name = pretty_name(g), legendgroup = g, showlegend = !direct, hoverinfo = "text",
                           text = paste0("<b>", pretty_name(g), "</b><br>PS index: ", round(f$ps_index, 2),
                                         "<br>Loess VST: ", round(f$fit, 2), " (n = ", f$n_libraries, ")"))
    }
    if (direct) {
      ends <- fits %>% group_by(group) %>% filter(ps_index == min(ps_index)) %>% ungroup()
      ends$y_lab <- spread_labels(ends$fit, diff(y_rng) * 0.075)
      for (i in seq_len(nrow(ends))) {
        g <- ends$group[i]
        focus <- is.null(hl) || g == hl
        annotations[[length(annotations) + 1]] <- list(
          x = ends$ps_index[i] - 0.015, y = ends$y_lab[i], xref = "x", yref = "y", xanchor = "left",
          text = group_label(g, group_var), showarrow = FALSE,
          font = list(size = 11, color = if (focus) colour_of(g) else "#aaaaaa"))
      }
    }
  }
  layout(p,
         xaxis = axis_style(title = list(text = "Photosynthetic index"), autorange = FALSE,
                            range = c(1.03, if (direct) -0.34 else -0.03),
                            tickvals = c(1, 0.75, 0.5, 0.25, 0),
                            ticktext = c("1<br><span style='color:#999'>photosynthetic</span>", "0.75", "0.50", "0.25",
                                         "0<br><span style='color:#999'>senescent</span>")),
         yaxis = axis_style(title = list(text = "Expression (VST)"), range = y_rng),
         shapes = ps_bands(), annotations = annotations, hovermode = "closest",
         plot_bgcolor = "rgba(0,0,0,0)", paper_bgcolor = "rgba(0,0,0,0)",
         legend = list(orientation = "h", yanchor = "top", y = -0.22, font = list(size = 11)),
         margin = list(t = 20, r = 10, b = 10, l = 55)) %>%
    plot_style(file_stub)
}

make_stripplot <- function(df, group_var, y_rng, violin = FALSE, highlight = "", file_stub = "distribution") {
  pal <- palette_for(group_var)
  df$group <- as.character(df[[group_var]])
  df$block <- if (group_var == "season") "all" else df$season
  slots <- df %>% group_by(block, group) %>%
    summarise(med = median(expression), n = n(), .groups = "drop") %>% arrange(block, group)
  blocks <- unique(slots$block)
  slots$x <- seq_len(nrow(slots)) + (match(slots$block, blocks) - 1) * 0.8
  hl <- if (!is.null(highlight) && nzchar(highlight) && highlight %in% slots$group) highlight else NULL
  set.seed(42)
  df <- df %>% left_join(slots %>% select(block, group, x), by = c("block", "group")) %>%
    mutate(xj = x + runif(n(), -0.22, 0.22))

  p <- plot_ly()
  for (i in seq_len(nrow(slots))) {
    sl <- slots[i, ]
    pts <- df[df$block == sl$block & df$group == sl$group, ]
    base <- if (sl$group %in% names(pal)) unname(pal[sl$group]) else "#555555"
    focus <- is.null(hl) || sl$group == hl
    col <- if (focus) base else GREY_CTX
    if (violin && nrow(pts) > 2) {
      p <- p %>% add_trace(type = "violin", x0 = sl$x, y = pts$expression, width = 0.85, points = FALSE,
                           fillcolor = col, opacity = 0.18, line = list(width = 0), spanmode = "hard",
                           hoverinfo = "skip", showlegend = FALSE, box = list(visible = FALSE),
                           meanline = list(visible = FALSE))
    }
    p <- p %>%
      add_trace(type = "scatter", mode = "markers", x = pts$xj, y = pts$expression,
                marker = list(color = col, size = 5, opacity = if (focus) 0.5 else 0.35), showlegend = FALSE,
                hoverinfo = "text",
                text = paste0(pts$sample_id, "<br>", pretty_name(pts$accession), " · ", pts$season,
                              " · week ", pts$week, "<br>VST: ", round(pts$expression, 2))) %>%
      add_trace(type = "scatter", mode = "lines", x = c(sl$x - 0.3, sl$x + 0.3), y = c(sl$med, sl$med),
                line = list(color = if (focus) "#222222" else "#9a9a9a", width = 3.5), showlegend = FALSE,
                hoverinfo = "text", text = paste0("<b>", pretty_name(sl$group), "</b><br>Median VST: ",
                                                  round(sl$med, 2), " (n = ", sl$n, ")"))
  }

  shapes <- list(); annotations <- list()
  if (length(blocks) > 1 || blocks[1] != "all") for (b in blocks) {
    xs <- slots$x[slots$block == b]
    annotations[[length(annotations) + 1]] <- list(x = mean(xs), y = 1, xref = "x", yref = "paper", yanchor = "bottom",
                                                   text = paste(b, "season"), showarrow = FALSE,
                                                   font = list(size = 12, color = "#555"))
    if (b != blocks[1]) shapes[[length(shapes) + 1]] <- list(type = "line", xref = "x", yref = "paper",
                                                             x0 = min(xs) - 0.9, x1 = min(xs) - 0.9, y0 = 0, y1 = 1,
                                                             line = list(color = "#dddddd", width = 1))
  }
  ticks <- sapply(slots$group, group_label, group_var = group_var)
  if (nrow(slots) <= 4) ticks <- paste0(ticks, "<br><span style='color:#999;font-size:10px'>n = ", slots$n, "</span>")
  p %>%
    layout(xaxis = axis_style(title = "", tickvals = slots$x, ticktext = ticks, automargin = TRUE,
                              tickangle = if (nrow(slots) > 4) -35 else 0, range = c(0.4, max(slots$x) + 0.6)),
           yaxis = axis_style(title = list(text = "Expression (VST)"), range = y_rng),
           shapes = shapes, annotations = annotations, hovermode = "closest",
           plot_bgcolor = "rgba(0,0,0,0)", paper_bgcolor = "rgba(0,0,0,0)",
           margin = list(t = 28, r = 10, b = 10, l = 55)) %>%
    plot_style(file_stub)
}

network_server <- function(input, output, session, prefix, net, root, go_expr, expr_tissues) {
  id <- function(x) paste0(prefix, x)
  src <- paste0(prefix, "view")
  umap <- net$umap

  updateSelectInput(session, id("module"),
                    choices = c("All modules" = "All", setNames(net$modules$module_label, net$modules$module_label)))

  query_og <- reactive({
    q <- str_trim(input[[id("query")]] %||% "")
    if (q == "") return(NULL)
    if (q %in% umap$gene) return(q)
    og <- unname(DATA$gene2og[q])
    if (length(og) && !is.na(og)) og else q
  }) %>% debounce(400)

  clicked <- reactiveVal(NULL)
  observeEvent(event_data("plotly_click", source = src), {
    clicked(event_data("plotly_click", source = src)$key)
    updateTextInput(session, id("query"), value = "")
  })
  observeEvent(input[[id("key_pick")]], {
    clicked(NULL)
    updateTextInput(session, id("query"), value = "")
    updateSelectInput(session, id("module"), selected = input[[id("key_pick")]])
  })

  active_gene <- reactive({
    q <- query_og()
    if (!is.null(q) && str_trim(input[[id("query")]] %||% "") != "") q else clicked()
  })

  output[[id("status")]] <- renderUI({
    q <- query_og()
    if (is.null(q) || str_trim(input[[id("query")]] %||% "") == "") return(NULL)
    if (q %in% umap$gene) {
      tags$div(paste("Highlighted:", q), style = "color: #2a7a2a; font-weight: bold; margin-top: 8px;")
    } else {
      tags$div("Not found in this network", style = "color: #c9302c; font-weight: bold; margin-top: 8px;")
    }
  })

  output[[id("key")]] <- renderUI({
    div(class = "module-key",
        lapply(seq_len(nrow(net$modules)), function(i) {
          m <- net$modules[i, ]
          light <- sum(col2rgb(m$color) * c(0.299, 0.587, 0.114)) > 150
          tags$span(style = paste0("background:", m$color, ";", if (light) " color:#222; text-shadow:none;" else ""),
                    onclick = sprintf("Shiny.setInputValue('%s', '%s', {priority: 'event'})", id("key_pick"), m$module_label),
                    title = "Highlight this module", m$module_label)
        }))
  })

  output[[id("plot")]] <- renderPlotly({
    plot_data <- umap
    edge_data <- net$edges
    g <- active_gene()
    mod <- input[[id("module")]] %||% "All"
    found <- !is.null(g) && g %in% plot_data$gene

    if (!is.null(g) && !found) {
      plot_data <- plot_data %>% mutate(display_color = "lightgrey", display_opacity = 0.15, display_size = 3)
      edge_data$hi <- FALSE
    } else if (found) {
      plot_data <- plot_data %>%
        mutate(display_color = ifelse(gene == g, color, "lightgrey"),
               display_opacity = ifelse(gene == g, 1, 0.15),
               display_size = ifelse(gene == g, 20, pmax(kME * 10, 3))) %>%
        arrange(gene == g)
      edge_data$hi <- edge_data$from == g | edge_data$to == g
    } else if (mod != "All") {
      plot_data <- plot_data %>%
        mutate(display_color = ifelse(module_label == mod, color, "lightgrey"),
               display_opacity = ifelse(module_label == mod, 0.9, 0.15),
               display_size = pmax(kME * 15, 3))
      edge_data$hi <- FALSE
    } else {
      plot_data <- plot_data %>% mutate(display_color = color, display_opacity = 0.8, display_size = pmax(kME * 15, 3))
      edge_data$hi <- FALSE
    }

    p <- plot_ly(source = src)
    bg <- edge_data[!edge_data$hi, ]
    hi <- edge_data[edge_data$hi, ]
    if (nrow(bg) > 0) p <- p %>% add_segments(data = bg, x = ~x0, xend = ~x1, y = ~y0, yend = ~y1,
                                               line = list(color = "lightgrey", width = 0.5), opacity = 0.08,
                                               hoverinfo = "none", showlegend = FALSE)
    if (nrow(hi) > 0) p <- p %>% add_segments(data = hi, x = ~x0, xend = ~x1, y = ~y0, yend = ~y1,
                                               line = list(color = "red", width = 2), opacity = 0.85,
                                               hoverinfo = "none", showlegend = FALSE)
    p %>%
      add_trace(data = plot_data, x = ~UMAP1, y = ~UMAP2, type = "scatter", mode = "markers",
                marker = list(color = ~display_color, size = ~display_size, opacity = ~display_opacity,
                              line = list(color = "white", width = 0.5)),
                text = ~paste0("OG: ", gene, "<br>Zea mays: ", Zea_mays_genes, "<br>Module: ", module_label,
                               "<br>kME: ", round(kME, 3)),
                hoverinfo = "text", key = ~gene, showlegend = FALSE) %>%
      layout(xaxis = list(title = "", showgrid = FALSE, zeroline = FALSE, showticklabels = FALSE),
             yaxis = list(title = "", showgrid = FALSE, zeroline = FALSE, showticklabels = FALSE),
             hovermode = "closest", margin = list(l = 5, r = 5, t = 5, b = 5),
             font = list(family = MONO, size = 11), hoverlabel = list(font = list(family = MONO, size = 11))) %>%
      event_register("plotly_click")
  })

  output[[id("details")]] <- renderUI({
    g <- active_gene()
    if (is.null(g) || !(g %in% umap$gene)) {
      return(p("Click a node or search for a gene to see its module, annotation and top co-expression partners.",
               style = "color: #777;"))
    }
    mod <- umap$module_label[umap$gene == g][1]
    tagList(
      div(class = "d-flex justify-content-between align-items-center mb-2",
          h6(paste0("Module eigengene pattern: ", mod), class = "mb-0"),
          actionButton(id("to_expr"), "Plot in Expression Explorer", class = "btn-sm btn-primary")),
      tags$img(src = paste0(root, net$plot_dir, "/module_", mod, ".png"),
               alt = paste("No plot available for module", mod),
               style = "width:100%; max-width:480px; border-radius:6px; margin-bottom:10px;"),
      hr(),
      h6("Selected node:")
    )
  })

  observeEvent(input[[id("to_expr")]], {
    g <- active_gene()
    if (!is.null(g)) go_expr(g, expr_tissues)
  })

  output[[id("node_table")]] <- renderTable({
    g <- active_gene()
    req(!is.null(g), g %in% umap$gene)
    u <- umap[umap$gene == g, ][1, ]
    zm <- strsplit(u$Zea_mays_genes, ", ")[[1]]
    d <- net$desc[match(zm, net$desc$gene_id_Zm), ]
    out <- data.frame(
      Orthogroup = g,
      "Zea mays gene" = ifelse(zm == "None", "None",
                               sprintf('<a href="https://www.maizegdb.org/gene_center/gene/%s" target="_blank">%s</a>', zm, zm)),
      Module = u$module_label,
      "Sink/source" = ifelse(is.na(d$sink_source), "—", d$sink_source),
      "kME (hub score)" = round(u$kME, 3),
      Description = ifelse(is.na(d$Description) | d$Description == "", "No description available", d$Description),
      check.names = FALSE, stringsAsFactors = FALSE
    )
    if (all(is.na(net$desc$sink_source))) out[["Sink/source"]] <- NULL
    out
  }, digits = 3, sanitize.text.function = function(x) x)

  partners <- reactive({
    g <- active_gene()
    req(!is.null(g), g %in% umap$gene)
    net$edges %>%
      filter(from == g | to == g) %>%
      mutate(partner = ifelse(from == g, to, from)) %>%
      select(partner, weight) %>%
      arrange(desc(weight)) %>%
      left_join(net$og_desc, by = c("partner" = "OG")) %>%
      mutate(Description = ifelse(is.na(Description), "No description available", Description),
             Zea_mays_genes = ifelse(is.na(Zea_mays_genes), "None", Zea_mays_genes),
             weight = round(weight, 4)) %>%
      transmute("Partner OG" = partner, "TOM weight" = weight, "Zea mays gene" = Zea_mays_genes, Description)
  })
  output[[id("inter_head")]] <- renderUI({
    n <- nrow(partners())
    tagList(hr(), h6(if (n) "Top interactions:" else "No interactions above the edge threshold."))
  })
  output[[id("interactions")]] <- renderTable({
    d <- partners(); req(nrow(d) > 0); d
  }, digits = 4)
}

                       selected = "Zm00001eb319560", server = TRUE)

  lookup <- reactive({
    q <- str_trim(input$gene_id %||% "")
    if (q == "") return(list(og = NULL, msg = NULL))
    if (q %in% DATA$expr_index$OG) return(list(og = q, msg = paste0(q, " ✓")))
    og <- unname(DATA$gene2og[q])
    if (length(og) && !is.na(og)) {
      if (og %in% DATA$expr_index$OG) return(list(og = og, msg = paste("Mapped orthogroup:", og)))
      return(list(og = NULL, msg = paste0("Maps to ", og, ", which is not in the expression atlas (filtered out).")))
    }
    list(og = NULL, msg = "ID not found in the atlas.")
  })
  target_og <- reactive(lookup()$og)

  output$og_id_display <- renderUI({
    l <- lookup()
    if (is.null(l$msg)) return(NULL)
    ok <- !is.null(l$og)
    span(l$msg, style = paste0("color:", if (ok) "#282" else "#e44", "; font-weight:", if (ok) "bold" else "normal",
                               "; font-size: 0.85em; display: block; margin: -8px 0 10px;"))
  })

  # ---- Filters: season -> tissue -> species -> accessions ----
  seasons_sel <- reactive(if (identical(input$season, "both")) c("2023", "2024") else input$season)

  observeEvent(seasons_sel(), {
    ch <- tissue_choices(seasons_sel())
    keep <- intersect(input$filter_tissue %||% "leaf", unlist(ch))
    if (!length(keep)) keep <- unlist(ch)[1]
    freezeReactiveValue(input, "filter_tissue")
    updateSelectizeInput(session, "filter_tissue", choices = ch, selected = keep)
  })

  first_species <- TRUE
  observeEvent(list(seasons_sel(), input$filter_tissue), {
    sel <- if (first_species) {
      c("Andropogon_gerardii", "Zea_mays", "Tripsacum_dactyloides", "Sorghum_bicolor",
        "Sorghastrum_nutans", "Zea_diploperennis")
    } else isolate(input$filter_species) %||% character()
    first_species <<- FALSE
    freezeReactiveValue(input, "filter_species")
    updateSelectizeInput(session, "filter_species",
                         choices = species_choices(seasons_sel(), input$filter_tissue %||% character()),
                         selected = sel)
  }, ignoreNULL = FALSE)

  avail_accessions <- reactive({
    DATA$avail %>%
      filter(species %in% (input$filter_species %||% character()), season %in% seasons_sel(),
             tissue_type %in% (input$filter_tissue %||% character())) %>%
      pull(accession) %>% unique() %>% sort()
  })

  observeEvent(list(input$filter_species, seasons_sel(), input$filter_tissue), {
    accs <- avail_accessions()
    sp <- input$filter_species %||% character()
    keep <- intersect(isolate(acc_pref()), accs)   # user's intended picks that exist for this season/tissue
    need <- sp[!sp %in% DATA$acc2sp[keep]]          # selected species with no accession picked yet
    sel <- union(keep, default_accessions(need, accs))
    freezeReactiveValue(input, "filter_accession")
    updateSelectizeInput(session, "filter_accession", choices = accession_choices(accs), selected = sel)
  }, ignoreNULL = FALSE)

  # Remember what the user intends, including picks hidden by the current season/tissue,
  # so switching 2024 -> Both brings the 2023 accessions back.
  acc_pref <- reactiveVal(character())
  observeEvent(input$filter_accession, {
    shown <- isolate(avail_accessions())
    acc_pref(union(setdiff(isolate(acc_pref()), shown), input$filter_accession %||% character()))
  }, ignoreNULL = FALSE, ignoreInit = TRUE)

  observeEvent(input$acc_all, {
    updateSelectizeInput(session, "filter_accession", selected = avail_accessions())
  })
  observeEvent(input$acc_none, {
    updateSelectizeInput(session, "filter_accession", selected = character())
  })

  output$missing_note <- renderUI({
    sp <- input$filter_species %||% character()
    a <- DATA$avail %>% filter(season %in% seasons_sel(), tissue_type %in% (input$filter_tissue %||% character()))
    miss <- setdiff(sp, a$species)
    if (!length(miss)) return(NULL)
    txt <- sapply(miss, function(s) paste0(pretty_name(s), " (",
                                           season_tag(DATA$avail$season[DATA$avail$species == s]), " only)"))
    div(class = "wx-note", style = "margin-top: 6px;",
        tags$b("No samples for this season/tissue: "), paste(txt, collapse = "; "))
  })

  # ---- Data ----
  og_expr <- reactive({
    og <- target_og(); req(og)
    get_expression(og, root)
  })
  filters <- reactive(list(seasons = seasons_sel(), tissue = input$filter_tissue,
                           acc = input$filter_accession)) %>% debounce(300)
  expr_df <- reactive({
    f <- filters()
    ex <- og_expr()
    md %>%
      filter(season %in% f$seasons, tissue_type %in% (f$tissue %||% character()),
             accession %in% (f$acc %||% character())) %>%
      inner_join(ex, by = "sample_id")
  })
  y_rng <- reactive({
    r <- range(expr_df()$expression, na.rm = TRUE)
    b <- max(diff(r) * 0.1, 0.2)
    c(r[1] - b, r[2] + b)
  })
  wx <- reactive({
    if (!isTRUE(input$wx_on) || identical(input$x_axis, "ps")) return(NULL)
    weather_by_week(input$wx_var, input$wx_window) %>%
      semi_join(distinct(md, season, week), by = c("season", "week"))
  })
  seasons_present <- reactive(intersect(c("2023", "2024"), unique(expr_df()$season)))

  file_stub <- function(kind) paste0(target_og() %||% "atlas", "_", kind)

  output$expr_title <- renderUI({
    og <- target_og(); req(og)
    q <- str_trim(input$gene_id %||% "")
    gene <- if (nzchar(q) && q != og) paste0(q, " (", og, ")") else og
    tis <- paste(tissue_labels[input$filter_tissue %||% character()], collapse = ", ")
    gv <- c(species = "species", accession = "accession", tissue_type = "tissue", tissue = "organ",
            life_history = "life history", season = "season")[[input$group_var]]
    tagList(h5(gene),
            p(paste0(tis, " · ", if (identical(input$season, "both")) "2023 and 2024" else input$season,
                     " · coloured by ", gv, if (identical(input$x_axis, "ps")) " · x: photosynthetic index" else "",
                     if (!(input$highlight %||% "none") %in% c("", "none")) paste0(" · highlighting ", pretty_name(input$highlight)) else "")))
  })

  observeEvent(list(expr_df(), input$group_var), {
    g <- sort(unique(as.character(expr_df()[[input$group_var]])))
    keep <- if (isTRUE(input$highlight %in% g)) input$highlight else "none"
    updateSelectInput(session, "highlight", choices = c("None (show all in colour)" = "none", setNames(g, pretty_name(g))),
                      selected = keep)
  })

  output$box_plot <- renderPlotly({
    df <- expr_df()
    validate(need(nrow(df) > 0, "No samples match the current filters."))
    make_stripplot(df, input$group_var, y_rng(), isTRUE(input$violin), input$highlight, file_stub("distribution"))
  })

  output$tc_header <- renderText(
    if (identical(input$x_axis, "ps")) "Trajectory along the photosynthetic index (loess ± 95% CI)" else "Weekly trajectory (mean ± SE)")
  output$tc_footer <- renderText(
    if (identical(input$x_axis, "ps")) "Dots: libraries. Lines: loess fit per group (span 0.5) with 95% CI. Selected seasons are combined; the index aligns the photosynthesis-to-senescence transition across species." else
      "Week and first sampling date. Ribbon: mean ± SE. Background: July (green) to October (yellow).")
  output$tc_ps <- renderPlotly({
    df <- expr_df()
    validate(need(any(!is.na(df$ps_index)), "No photosynthetic index for these libraries."))
    make_ps_curve(df, input$group_var, y_rng(), input$highlight, file_stub("ps_index"))
  })
  output$tc_panels <- renderUI({
    if (identical(input$x_axis, "ps")) {
      validate(need(nrow(expr_df()) > 0, "No samples match the current filters."))
      return(tagList(h6("Selected seasons combined", class = "mt-2 mb-0"), plotlyOutput("tc_ps", height = "560px")))
    }
    s <- seasons_present()
    validate(need(length(s) > 0, "No samples match the current filters."))
    h <- if (length(s) == 1) "560px" else "420px"
    tagList(lapply(s, function(x) tagList(
      h6(paste(x, "season"), class = "mt-2 mb-0"),
      plotlyOutput(paste0("tc_", x), height = h))))
  })
  for (s in c("2023", "2024")) local({
    ss <- s
    output[[paste0("tc_", ss)]] <- renderPlotly({
      df <- expr_df()
      req(ss %in% df$season)
      make_timecourse(df, ss, input$group_var, y_rng(), wx(), input$wx_var, input$highlight,
                      file_stub(paste0("timecourse_", ss)))
    })
  })

  output$dl_samples <- downloadHandler(
    filename = function() paste0(file_stub("samples"), ".csv"),
    content = function(file) {
      d <- expr_df() %>%
        transmute(orthogroup = target_og(), sample_id, species, accession, season, tissue = tissue_type,
                  week, date = format(date), julian_day = julian_days, ps_index, vst_expression = expression) %>%
        arrange(season, species, accession, tissue, week)
      write.csv(d, file, row.names = FALSE)
    }
  )
  output$dl_summary <- downloadHandler(
    filename = function() paste0(file_stub(if (identical(input$x_axis, "ps")) "ps_index_loess" else "timecourse"), ".csv"),
    content = function(file) {
      gv <- input$group_var
      if (identical(input$x_axis, "ps")) {
        f <- ps_fits(expr_df() %>% filter(!is.na(ps_index)) %>% mutate(group = as.character(.data[[gv]])))
        f <- f %>% mutate(across(c(ps_index, fit, se, ci_low, ci_high), ~ round(.x, 4))) %>%
          rename(loess_vst = fit) %>% mutate(orthogroup = target_og(), grouped_by = gv, .before = 1)
        write.csv(f, file, row.names = FALSE)
        return(invisible())
      }
      wkd <- md %>% group_by(season, week) %>% summarise(first_sampling_date = format(min(date)), .groups = "drop")
      d <- expr_df() %>%
        group_by(season, group = as.character(.data[[gv]]), week) %>%
        summarise(mean_vst = round(mean(expression), 4),
                  se_vst = round(if (n() > 1) sd(expression) / sqrt(n()) else 0, 4),
                  median_vst = round(median(expression), 4), n = n(), .groups = "drop") %>%
        left_join(wkd, by = c("season", "week")) %>%
        mutate(orthogroup = target_og(), grouped_by = gv, .before = 1)
      w <- wx()
      if (!is.null(w)) {
        d <- d %>% left_join(w %>% rename(weather_value = value), by = c("season", "week")) %>%
          mutate(weather_variable = weather_axis[[input$wx_var]],
                 weather_window = if (input$wx_window == "calendar") "calendar week" else "7 days before sampling",
                 weather_value = round(weather_value, 2)) %>%
          relocate(weather_variable, weather_window, .before = weather_value)
      }
      write.csv(d %>% arrange(season, group, week), file, row.names = FALSE)
    }
  )

  # ---- Networks ----
  go_expr <- function(og, tissues) {
    updateSelectizeInput(session, "gene_id", choices = DATA$gene_choices, selected = og, server = TRUE)
    updateSelectizeInput(session, "filter_tissue", selected = intersect(tissues, unlist(tissue_choices(seasons_sel()))))
    nav_select("main_nav", "Expression Explorer", session = session)
  }
  network_server(input, output, session, "leaf_", DATA$leaf, root, go_expr, "leaf")
  network_server(input, output, session, "und_", DATA$und, root, go_expr,
                 c("root", "fine_roots", "thick_roots", "rhizome", "stalk"))
}
